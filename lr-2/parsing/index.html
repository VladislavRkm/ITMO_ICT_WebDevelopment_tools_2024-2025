    Парсинг веб-страниц

    Описание

    В этой части лабораторной работы мы реализуем парсер веб-страниц, используя три различных подхода к асинхронному программированию: asyncio, threading и multiprocessing. Это позволит нам сравнить производительность и особенности каждого подхода.

    Цели

    • Реализовать парсер с использованием asyncio
    • Реализовать парсер с использованием threading
    • Реализовать парсер с использованием multiprocessing
    • Сравнить производительность подходов
    • Понять различия в использовании ресурсов

    Асинхронный парсер (asyncio)

    async_parser.py

    import asyncio
import aiohttp
from bs4 import BeautifulSoup
import time
from typing import List, Dict
import logging

# Настройка логирования
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class AsyncWebParser:
    def __init__(self, max_concurrent: int = 10):
        self.max_concurrent = max_concurrent
        self.semaphore = asyncio.Semaphore(max_concurrent)
        self.session = None

    async def __aenter__(self):
        self.session = aiohttp.ClientSession()
        return self

    async def __aexit__(self, exc_type, exc_val, exc_tb):
        if self.session:
            await self.session.close()

    async def fetch_page(self, url: str) -> str:
        """Асинхронная загрузка страницы"""
        async with self.semaphore:
            try:
                async with self.session.get(url, timeout=30) as response:
                    if response.status == 200:
                        return await response.text()
                    else:
                        logger.warning(f"HTTP {response.status} for {url}")
                        return ""
            except Exception as e:
                logger.error(f"Error fetching {url}: {e}")
                return ""

    async def parse_page(self, url: str) -> Dict:
        """Парсинг одной страницы"""
        html = await self.fetch_page(url)
        if not html:
            return {"url": url, "title": "", "links": [], "error": "Failed to fetch"}

        try:
            soup = BeautifulSoup(html, 'html.parser')

            # Извлечение заголовка
            title = soup.find('title')
            title_text = title.get_text().strip() if title else ""

            # Извлечение ссылок
            links = []
            for link in soup.find_all('a', href=True):
                href = link['href']
                text = link.get_text().strip()
                if href.startswith('http'):
                    links.append({"url": href, "text": text})

            return {
                "url": url,
                "title": title_text,
                "links": links,
                "links_count": len(links)
            }

        except Exception as e:
            logger.error(f"Error parsing {url}: {e}")
            return {"url": url, "title": "", "links": [], "error": str(e)}

    async def parse_urls(self, urls: List[str]) -> List[Dict]:
        """Парсинг списка URL"""
        logger.info(f"Starting async parsing of {len(urls)} URLs")
        start_time = time.time()

        # Создание задач для всех URL
        tasks = [self.parse_page(url) for url in urls]

        # Выполнение всех задач параллельно
        results = await asyncio.gather(*tasks, return_exceptions=True)

        # Обработка результатов
        parsed_results = []
        for i, result in enumerate(results):
            if isinstance(result, Exception):
                logger.error(f"Task {i} failed: {result}")
                parsed_results.append({
                    "url": urls[i],
                    "title": "",
                    "links": [],
                    "error": str(result)
                })
            else:
                parsed_results.append(result)

        end_time = time.time()
        logger.info(f"Async parsing completed in {end_time - start_time:.2f} seconds")

        return parsed_results

async def main():
    # Список URL для парсинга
    urls = [
        "https://httpbin.org/html",
        "https://httpbin.org/json",
        "https://httpbin.org/xml",
        "https://httpbin.org/robots.txt",
        "https://httpbin.org/user-agent",
        "https://httpbin.org/headers",
        "https://httpbin.org/ip",
        "https://httpbin.org/uuid",
        "https://httpbin.org/delay/1",
        "https://httpbin.org/delay/2"
    ]

    # Создание и использование парсера
    async with AsyncWebParser(max_concurrent=5) as parser:
        results = await parser.parse_urls(urls)

        # Вывод результатов
        for result in results:
            print(f"URL: {result['url']}")
            print(f"Title: {result['title']}")
            print(f"Links: {result['links_count']}")
            if 'error' in result:
                print(f"Error: {result['error']}")
            print("-" * 50)

if __name__ == "__main__":
    asyncio.run(main())

    Особенности асинхронного подхода

    1. Кооперативная многозадачность - задачи добровольно уступают контроль
    2. Эффективное использование ресурсов - один поток обрабатывает множество соединений
    3. Простота отладки - последовательное выполнение в рамках одной задачи
    4. Встроенная обработка ошибок - return_exceptions=True в gather()

    Многопоточный парсер (threading)

    threading_parser.py

    import threading
import requests
from bs4 import BeautifulSoup
import time
from typing import List, Dict
import logging
from concurrent.futures import ThreadPoolExecutor, as_completed
from queue import Queue
import json

# Настройка логирования
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class ThreadingWebParser:
    def __init__(self, max_workers: int = 10):
        self.max_workers = max_workers
        self.session = requests.Session()
        self.results_queue = Queue()
        self.lock = threading.Lock()

    def fetch_page(self, url: str) -> str:
        """Загрузка страницы в отдельном потоке"""
        try:
            response = self.session.get(url, timeout=30)
            response.raise_for_status()
            return response.text
        except Exception as e:
            logger.error(f"Error fetching {url}: {e}")
            return ""

    def parse_page(self, url: str) -> Dict:
        """Парсинг одной страницы"""
        html = self.fetch_page(url)
        if not html:
            return {"url": url, "title": "", "links": [], "error": "Failed to fetch"}

        try:
            soup = BeautifulSoup(html, 'html.parser')

            # Извлечение заголовка
            title = soup.find('title')
            title_text = title.get_text().strip() if title else ""

            # Извлечение ссылок
            links = []
            for link in soup.find_all('a', href=True):
                href = link['href']
                text = link.get_text().strip()
                if href.startswith('http'):
                    links.append({"url": href, "text": text})

            result = {
                "url": url,
                "title": title_text,
                "links": links,
                "links_count": len(links)
            }

            # Потокобезопасное добавление результата
            with self.lock:
                self.results_queue.put(result)

            return result

        except Exception as e:
            logger.error(f"Error parsing {url}: {e}")
            error_result = {"url": url, "title": "", "links": [], "error": str(e)}
            with self.lock:
                self.results_queue.put(error_result)
            return error_result

    def parse_urls(self, urls: List[str]) -> List[Dict]:
        """Парсинг списка URL с использованием пула потоков"""
        logger.info(f"Starting threading parsing of {len(urls)} URLs")
        start_time = time.time()

        results = []

        # Использование ThreadPoolExecutor для управления потоками
        with ThreadPoolExecutor(max_workers=self.max_workers) as executor:
            # Отправка задач в пул
            future_to_url = {
                executor.submit(self.parse_page, url): url 
                for url in urls
            }

            # Сбор результатов по мере завершения
            for future in as_completed(future_to_url):
                url = future_to_url[future]
                try:
                    result = future.result()
                    results.append(result)
                    logger.info(f"Completed parsing {url}")
                except Exception as e:
                    logger.error(f"Task for {url} generated an exception: {e}")
                    results.append({
                        "url": url,
                        "title": "",
                        "links": [],
                        "error": str(e)
                    })

        end_time = time.time()
        logger.info(f"Threading parsing completed in {end_time - start_time:.2f} seconds")

        return results

    def parse_urls_with_queue(self, urls: List[str]) -> List[Dict]:
        """Альтернативная реализация с использованием очереди"""
        logger.info(f"Starting queue-based parsing of {len(urls)} URLs")
        start_time = time.time()

        # Очистка очереди
        while not self.results_queue.empty():
            self.results_queue.get()

        # Создание и запуск потоков
        threads = []
        for url in urls:
            thread = threading.Thread(target=self.parse_page, args=(url,))
            threads.append(thread)
            thread.start()

        # Ожидание завершения всех потоков
        for thread in threads:
            thread.join()

        # Сбор результатов из очереди
        results = []
        while not self.results_queue.empty():
            results.append(self.results_queue.get())

        end_time = time.time()
        logger.info(f"Queue-based parsing completed in {end_time - start_time:.2f} seconds")

        return results

def main():
    # Список URL для парсинга
    urls = [
        "https://httpbin.org/html",
        "https://httpbin.org/json",
        "https://httpbin.org/xml",
        "https://httpbin.org/robots.txt",
        "https://httpbin.org/user-agent",
        "https://httpbin.org/headers",
        "https://httpbin.org/ip",
        "https://httpbin.org/uuid",
        "https://httpbin.org/delay/1",
        "https://httpbin.org/delay/2"
    ]

    # Создание парсера
    parser = ThreadingWebParser(max_workers=5)

    # Парсинг с использованием ThreadPoolExecutor
    results = parser.parse_urls(urls)

    # Вывод результатов
    for result in results:
        print(f"URL: {result['url']}")
        print(f"Title: {result['title']}")
        print(f"Links: {result['links_count']}")
        if 'error' in result:
            print(f"Error: {result['error']}")
        print("-" * 50)

if __name__ == "__main__":
    main()

    Особенности многопоточного подхода

    1. Параллельное выполнение - потоки выполняются одновременно
    2. Общая память - потоки могут обмениваться данными
    3. GIL ограничения - только один поток выполняет Python код одновременно
    4. Синхронизация - необходимость в locks и queues

    Многопроцессный парсер (multiprocessing)

    multiprocessing_parser.py

    import multiprocessing
import requests
from bs4 import BeautifulSoup
import time
from typing import List, Dict
import logging
from concurrent.futures import ProcessPoolExecutor, as_completed
import json
import os

# Настройка логирования
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

def fetch_page(url: str) -> str:
    """Загрузка страницы в отдельном процессе"""
    try:
        session = requests.Session()
        response = session.get(url, timeout=30)
        response.raise_for_status()
        return response.text
    except Exception as e:
        logger.error(f"Error fetching {url}: {e}")
        return ""

def parse_page(url: str) -> Dict:
    """Парсинг одной страницы в отдельном процессе"""
    html = fetch_page(url)
    if not html:
        return {"url": url, "title": "", "links": [], "error": "Failed to fetch"}

    try:
        soup = BeautifulSoup(html, 'html.parser')

        # Извлечение заголовка
        title = soup.find('title')
        title_text = title.get_text().strip() if title else ""

        # Извлечение ссылок
        links = []
        for link in soup.find_all('a', href=True):
            href = link['href']
            text = link.get_text().strip()
            if href.startswith('http'):
                links.append({"url": href, "text": text})

        return {
            "url": url,
            "title": title_text,
            "links": links,
            "links_count": len(links),
            "process_id": os.getpid()
        }

    except Exception as e:
        logger.error(f"Error parsing {url}: {e}")
        return {"url": url, "title": "", "links": [], "error": str(e), "process_id": os.getpid()}

class MultiprocessingWebParser:
    def __init__(self, max_workers: int = None):
        self.max_workers = max_workers or min(32, os.cpu_count() + 4)

    def parse_urls(self, urls: List[str]) -> List[Dict]:
        """Парсинг списка URL с использованием пула процессов"""
        logger.info(f"Starting multiprocessing parsing of {len(urls)} URLs")
        logger.info(f"Using {self.max_workers} processes")
        start_time = time.time()

        results = []

        # Использование ProcessPoolExecutor для управления процессами
        with ProcessPoolExecutor(max_workers=self.max_workers) as executor:
            # Отправка задач в пул
            future_to_url = {
                executor.submit(parse_page, url): url 
                for url in urls
            }

            # Сбор результатов по мере завершения
            for future in as_completed(future_to_url):
                url = future_to_url[future]
                try:
                    result = future.result()
                    results.append(result)
                    logger.info(f"Completed parsing {url} in process {result.get('process_id', 'unknown')}")
                except Exception as e:
                    logger.error(f"Task for {url} generated an exception: {e}")
                    results.append({
                        "url": url,
                        "title": "",
                        "links": [],
                        "error": str(e)
                    })

        end_time = time.time()
        logger.info(f"Multiprocessing parsing completed in {end_time - start_time:.2f} seconds")

        return results

    def parse_urls_chunked(self, urls: List[str], chunk_size: int = 5) -> List[Dict]:
        """Парсинг URL блоками для лучшего контроля памяти"""
        logger.info(f"Starting chunked parsing of {len(urls)} URLs")
        start_time = time.time()

        all_results = []

        # Разделение URL на блоки
        url_chunks = [urls[i:i + chunk_size] for i in range(0, len(urls), chunk_size)]

        with ProcessPoolExecutor(max_workers=self.max_workers) as executor:
            for chunk in url_chunks:
                logger.info(f"Processing chunk of {len(chunk)} URLs")

                # Отправка блока задач
                future_to_url = {
                    executor.submit(parse_page, url): url 
                    for url in chunk
                }

                # Сбор результатов блока
                chunk_results = []
                for future in as_completed(future_to_url):
                    url = future_to_url[future]
                    try:
                        result = future.result()
                        chunk_results.append(result)
                    except Exception as e:
                        logger.error(f"Task for {url} generated an exception: {e}")
                        chunk_results.append({
                            "url": url,
                            "title": "",
                            "links": [],
                            "error": str(e)
                        })

                all_results.extend(chunk_results)

        end_time = time.time()
        logger.info(f"Chunked parsing completed in {end_time - start_time:.2f} seconds")

        return all_results

def parse_urls_batch(urls_batch: List[str]) -> List[Dict]:
    """Функция для пакетной обработки URL"""
    results = []
    for url in urls_batch:
        result = parse_page(url)
        results.append(result)
    return results

def main():
    # Список URL для парсинга
    urls = [
        "https://httpbin.org/html",
        "https://httpbin.org/json",
        "https://httpbin.org/xml",
        "https://httpbin.org/robots.txt",
        "https://httpbin.org/user-agent",
        "https://httpbin.org/headers",
        "https://httpbin.org/ip",
        "https://httpbin.org/uuid",
        "https://httpbin.org/delay/1",
        "https://httpbin.org/delay/2"
    ]

    # Создание парсера
    parser = MultiprocessingWebParser(max_workers=4)

    # Парсинг с использованием ProcessPoolExecutor
    results = parser.parse_urls(urls)

    # Вывод результатов
    for result in results:
        print(f"URL: {result['url']}")
        print(f"Title: {result['title']}")
        print(f"Links: {result['links_count']}")
        print(f"Process ID: {result.get('process_id', 'N/A')}")
        if 'error' in result:
            print(f"Error: {result['error']}")
        print("-" * 50)

if __name__ == "__main__":
    # Важно для multiprocessing на Windows
    multiprocessing.freeze_support()
    main()

    Особенности многопроцессного подхода

    1. Истинный параллелизм - процессы выполняются на разных ядрах CPU
    2. Изоляция процессов - каждый процесс имеет свою память
    3. Высокие накладные расходы - создание процессов дороже потоков
    4. Сложность обмена данными - необходимо сериализация

    Сравнение производительности

    Скрипт для тестирования

    import time
import asyncio
import statistics
from typing import List, Dict
import psutil
import os

# Импорт парсеров
from async_parser import AsyncWebParser
from threading_parser import ThreadingWebParser
from multiprocessing_parser import MultiprocessingWebParser

def measure_performance(parser_func, urls: List[str], name: str) -> Dict:
    """Измерение производительности парсера"""
    print(f"\nТестирование {name}...")

    # Измерение времени
    start_time = time.time()
    start_memory = psutil.Process().memory_info().rss / 1024 / 1024  # MB

    # Выполнение парсинга
    if name == "Async":
        results = asyncio.run(parser_func(urls))
    else:
        results = parser_func(urls)

    end_time = time.time()
    end_memory = psutil.Process().memory_info().rss / 1024 / 1024  # MB

    execution_time = end_time - start_time
    memory_usage = end_memory - start_memory

    # Статистика результатов
    successful = sum(1 for r in results if 'error' not in r)
    failed = len(results) - successful

    return {
        "name": name,
        "execution_time": execution_time,
        "memory_usage": memory_usage,
        "successful": successful,
        "failed": failed,
        "total": len(results),
        "urls_per_second": len(urls) / execution_time
    }

    Обработка ошибок и устойчивость

    Общие принципы

    1. Таймауты - ограничение времени ожидания
    2. Повторные попытки - retry механизм
    3. Логирование - детальное логирование ошибок
    4. Graceful degradation - продолжение работы при ошибках

    Реализация retry механизма

    import asyncio
import time
from functools import wraps

def async_retry(max_retries=3, delay=1):
    """Декоратор для повторных попыток в asyncio"""
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            last_exception = None
            for attempt in range(max_retries):
                try:
                    return await func(*args, **kwargs)
                except Exception as e:
                    last_exception = e
                    if attempt < max_retries - 1:
                        await asyncio.sleep(delay * (2 ** attempt))
            raise last_exception
        return wrapper
    return decorator

def sync_retry(max_retries=3, delay=1):
    """Декоратор для повторных попыток в синхронном коде"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            last_exception = None
            for attempt in range(max_retries):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    last_exception = e
                    if attempt < max_retries - 1:
                        time.sleep(delay * (2 ** attempt))
            raise last_exception
        return wrapper
    return decorator

    Интеграция с веб-фреймворками

    # FastAPI endpoint для парсинга
from fastapi import FastAPI, BackgroundTasks
import asyncio

app = FastAPI()

@app.post("/parse")
async def parse_urls(urls: List[str], background_tasks: BackgroundTasks):
    async with AsyncWebParser() as parser:
        results = await parser.parse_urls(urls)
    return {"results": results, "count": len(results)}

@app.post("/parse/background")
async def parse_urls_background(urls: List[str], background_tasks: BackgroundTasks):
    async def parse_task():
        async with AsyncWebParser() as parser:
            return await parser.parse_urls(urls)

    background_tasks.add_task(parse_task)
    return {"message": "Parsing started in background"}

    Заключение

    Парсинг веб-страниц демонстрирует различные подходы к асинхронному программированию:

    • asyncio - лучший выбор для I/O-интенсивных задач парсинга
    • threading - простой способ добавить параллелизм
    • multiprocessing - избыточен для сетевых операций

    Выбор подхода зависит от конкретных требований: производительность, простота разработки, масштабируемость.
