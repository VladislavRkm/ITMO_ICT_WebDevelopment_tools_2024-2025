    Подсчет суммы

    Описание

    В этой части лабораторной работы мы реализуем алгоритм параллельного подсчета суммы больших массивов данных, используя три различных подхода: asyncio, threading и multiprocessing. Это позволит нам изучить эффективность каждого подхода для CPU-интенсивных задач.

    Цели

    • Реализовать параллельный подсчет суммы с использованием asyncio
    • Реализовать параллельный подсчет суммы с использованием threading
    • Реализовать параллельный подсчет суммы с использованием multiprocessing
    • Сравнить производительность подходов для CPU-интенсивных задач
    • Понять различия в обработке данных

    Асинхронный подсчет (asyncio)

    sum_count_async.py

    import asyncio
import time
import random
from typing import List, Tuple
import logging

# Настройка логирования
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class AsyncSumCalculator:
    def __init__(self, chunk_size: int = 10000):
        self.chunk_size = chunk_size

    def generate_data(self, size: int) -> List[int]:
        """Генерация тестовых данных"""
        logger.info(f"Generating {size} random numbers...")
        return [random.randint(1, 1000) for _ in range(size)]

    def split_data(self, data: List[int]) -> List[List[int]]:
        """Разделение данных на блоки"""
        chunks = []
        for i in range(0, len(data), self.chunk_size):
            chunk = data[i:i + self.chunk_size]
            chunks.append(chunk)
        logger.info(f"Split data into {len(chunks)} chunks")
        return chunks

    async def calculate_chunk_sum(self, chunk: List[int], chunk_id: int) -> Tuple[int, int]:
        """Асинхронный подсчет суммы для одного блока"""
        logger.info(f"Starting calculation for chunk {chunk_id} with {len(chunk)} elements")

        # Имитация CPU-интенсивной работы
        await asyncio.sleep(0.1)

        # Подсчет суммы
        chunk_sum = sum(chunk)

        logger.info(f"Chunk {chunk_id} completed: sum = {chunk_sum}")
        return chunk_id, chunk_sum

    async def calculate_sum_async(self, data: List[int]) -> int:
        """Асинхронный подсчет общей суммы"""
        logger.info(f"Starting async sum calculation for {len(data)} elements")
        start_time = time.time()

        # Разделение данных на блоки
        chunks = self.split_data(data)

        # Создание задач для всех блоков
        tasks = [
            self.calculate_chunk_sum(chunk, i) 
            for i, chunk in enumerate(chunks)
        ]

        # Выполнение всех задач параллельно
        results = await asyncio.gather(*tasks)

        # Подсчет общей суммы
        total_sum = sum(chunk_sum for _, chunk_sum in results)

        end_time = time.time()
        execution_time = end_time - start_time

        logger.info(f"Async calculation completed in {execution_time:.2f} seconds")
        logger.info(f"Total sum: {total_sum}")

        return total_sum

    async def calculate_sum_with_semaphore(self, data: List[int], max_concurrent: int = 4) -> int:
        """Асинхронный подсчет с ограничением конкурентности"""
        logger.info(f"Starting async sum calculation with semaphore for {len(data)} elements")
        start_time = time.time()

        # Создание семафора для ограничения конкурентности
        semaphore = asyncio.Semaphore(max_concurrent)

        async def calculate_chunk_with_semaphore(chunk: List[int], chunk_id: int) -> Tuple[int, int]:
            async with semaphore:
                return await self.calculate_chunk_sum(chunk, chunk_id)

        # Разделение данных на блоки
        chunks = self.split_data(data)

        # Создание задач с семафором
        tasks = [
            calculate_chunk_with_semaphore(chunk, i) 
            for i, chunk in enumerate(chunks)
        ]

        # Выполнение задач
        results = await asyncio.gather(*tasks)

        # Подсчет общей суммы
        total_sum = sum(chunk_sum for _, chunk_sum in results)

        end_time = time.time()
        execution_time = end_time - start_time

        logger.info(f"Async calculation with semaphore completed in {execution_time:.2f} seconds")
        logger.info(f"Total sum: {total_sum}")

        return total_sum

async def main():
    # Создание калькулятора
    calculator = AsyncSumCalculator(chunk_size=50000)

    # Генерация тестовых данных
    data_size = 1000000  # 1 миллион элементов
    data = calculator.generate_data(data_size)

    print(f"Generated {len(data)} random numbers")
    print(f"Expected sum (approximate): {len(data) * 500}")  # Среднее значение 500

    # Тестирование обычного асинхронного подсчета
    print("\n" + "="*50)
    print("ASYNC CALCULATION")
    print("="*50)

    result1 = await calculator.calculate_sum_async(data)

    # Тестирование асинхронного подсчета с семафором
    print("\n" + "="*50)
    print("ASYNC CALCULATION WITH SEMAPHORE")
    print("="*50)

    result2 = await calculator.calculate_sum_with_semaphore(data, max_concurrent=4)

    # Проверка результатов
    print(f"\nResults match: {result1 == result2}")
    print(f"Final sum: {result1}")

if __name__ == "__main__":
    asyncio.run(main())

    Особенности асинхронного подхода

    1. Кооперативная многозадачность - задачи добровольно уступают контроль
    2. Ограниченная эффективность для CPU - GIL ограничивает параллелизм
    3. Простота реализации - легко добавлять новые задачи
    4. Контроль конкурентности - семафоры для ограничения нагрузки

    Многопоточный подсчет (threading)

    sum_count_threading.py

    import threading
import time
import random
from typing import List, Tuple
import logging
from concurrent.futures import ThreadPoolExecutor, as_completed
from queue import Queue

# Настройка логирования
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class ThreadingSumCalculator:
    def __init__(self, chunk_size: int = 10000):
        self.chunk_size = chunk_size
        self.results_queue = Queue()
        self.lock = threading.Lock()

    def generate_data(self, size: int) -> List[int]:
        """Генерация тестовых данных"""
        logger.info(f"Generating {size} random numbers...")
        return [random.randint(1, 1000) for _ in range(size)]

    def split_data(self, data: List[int]) -> List[List[int]]:
        """Разделение данных на блоки"""
        chunks = []
        for i in range(0, len(data), self.chunk_size):
            chunk = data[i:i + self.chunk_size]
            chunks.append(chunk)
        logger.info(f"Split data into {len(chunks)} chunks")
        return chunks

    def calculate_chunk_sum(self, chunk: List[int], chunk_id: int) -> Tuple[int, int]:
        """Подсчет суммы для одного блока в отдельном потоке"""
        logger.info(f"Starting calculation for chunk {chunk_id} with {len(chunk)} elements")

        # Имитация CPU-интенсивной работы
        time.sleep(0.1)

        # Подсчет суммы
        chunk_sum = sum(chunk)

        logger.info(f"Chunk {chunk_id} completed: sum = {chunk_sum}")
        return chunk_id, chunk_sum

    def calculate_sum_threading(self, data: List[int], max_workers: int = 4) -> int:
        """Многопоточный подсчет общей суммы"""
        logger.info(f"Starting threading sum calculation for {len(data)} elements")
        start_time = time.time()

        # Разделение данных на блоки
        chunks = self.split_data(data)

        results = []

        # Использование ThreadPoolExecutor
        with ThreadPoolExecutor(max_workers=max_workers) as executor:
            # Отправка задач в пул
            future_to_chunk = {
                executor.submit(self.calculate_chunk_sum, chunk, i): i 
                for i, chunk in enumerate(chunks)
            }

            # Сбор результатов по мере завершения
            for future in as_completed(future_to_chunk):
                chunk_id = future_to_chunk[future]
                try:
                    result = future.result()
                    results.append(result)
                    logger.info(f"Completed chunk {chunk_id}")
                except Exception as e:
                    logger.error(f"Chunk {chunk_id} generated an exception: {e}")

        # Подсчет общей суммы
        total_sum = sum(chunk_sum for _, chunk_sum in results)

        end_time = time.time()
        execution_time = end_time - start_time

        logger.info(f"Threading calculation completed in {execution_time:.2f} seconds")
        logger.info(f"Total sum: {total_sum}")

        return total_sum

    def calculate_sum_with_queue(self, data: List[int]) -> int:
        """Альтернативная реализация с использованием очереди"""
        logger.info(f"Starting queue-based sum calculation for {len(data)} elements")
        start_time = time.time()

        # Очистка очереди
        while not self.results_queue.empty():
            self.results_queue.get()

        # Разделение данных на блоки
        chunks = self.split_data(data)

        # Создание и запуск потоков
        threads = []
        for i, chunk in enumerate(chunks):
            thread = threading.Thread(
                target=self._calculate_chunk_with_queue, 
                args=(chunk, i)
            )
            threads.append(thread)
            thread.start()

        # Ожидание завершения всех потоков
        for thread in threads:
            thread.join()

        # Сбор результатов из очереди
        results = []
        while not self.results_queue.empty():
            results.append(self.results_queue.get())

        # Подсчет общей суммы
        total_sum = sum(chunk_sum for _, chunk_sum in results)

        end_time = time.time()
        execution_time = end_time - start_time

        logger.info(f"Queue-based calculation completed in {execution_time:.2f} seconds")
        logger.info(f"Total sum: {total_sum}")

        return total_sum

    def _calculate_chunk_with_queue(self, chunk: List[int], chunk_id: int):
        """Вспомогательная функция для работы с очередью"""
        result = self.calculate_chunk_sum(chunk, chunk_id)
        with self.lock:
            self.results_queue.put(result)

def main():
    # Создание калькулятора
    calculator = ThreadingSumCalculator(chunk_size=50000)

    # Генерация тестовых данных
    data_size = 1000000  # 1 миллион элементов
    data = calculator.generate_data(data_size)

    print(f"Generated {len(data)} random numbers")
    print(f"Expected sum (approximate): {len(data) * 500}")  # Среднее значение 500

    # Тестирование многопоточного подсчета
    print("\n" + "="*50)
    print("THREADING CALCULATION")
    print("="*50)

    result1 = calculator.calculate_sum_threading(data, max_workers=4)

    # Тестирование подсчета с очередью
    print("\n" + "="*50)
    print("QUEUE-BASED CALCULATION")
    print("="*50)

    result2 = calculator.calculate_sum_with_queue(data)

    # Проверка результатов
    print(f"\nResults match: {result1 == result2}")
    print(f"Final sum: {result1}")

if __name__ == "__main__":
    main()

    Особенности многопоточного подхода

    1. Параллельное выполнение - потоки выполняются одновременно
    2. GIL ограничения - только один поток выполняет Python код одновременно
    3. Общая память - потоки могут обмениваться данными
    4. Синхронизация - необходимость в locks и queues

    Многопроцессный подсчет (multiprocessing)

    sum count_multiprocessing.py

    import multiprocessing
import time
import random
from typing import List, Tuple
import logging
from concurrent.futures import ProcessPoolExecutor, as_completed
import os

# Настройка логирования
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

def generate_data(size: int) -> List[int]:
    """Генерация тестовых данных"""
    logger.info(f"Generating {size} random numbers...")
    return [random.randint(1, 1000) for _ in range(size)]

def split_data(data: List[int], chunk_size: int) -> List[List[int]]:
    """Разделение данных на блоки"""
    chunks = []
    for i in range(0, len(data), chunk_size):
        chunk = data[i:i + chunk_size]
        chunks.append(chunk)
    logger.info(f"Split data into {len(chunks)} chunks")
    return chunks

def calculate_chunk_sum(chunk_data: Tuple[List[int], int]) -> Tuple[int, int]:
    """Подсчет суммы для одного блока в отдельном процессе"""
    chunk, chunk_id = chunk_data

    logger.info(f"Starting calculation for chunk {chunk_id} with {len(chunk)} elements in process {os.getpid()}")

    # Имитация CPU-интенсивной работы
    time.sleep(0.1)

    # Подсчет суммы
    chunk_sum = sum(chunk)

    logger.info(f"Chunk {chunk_id} completed: sum = {chunk_sum} in process {os.getpid()}")
    return chunk_id, chunk_sum

class MultiprocessingSumCalculator:
    def __init__(self, chunk_size: int = 10000):
        self.chunk_size = chunk_size

    def calculate_sum_multiprocessing(self, data: List[int], max_workers: int = None) -> int:
        """Многопроцессный подсчет общей суммы"""
        if max_workers is None:
            max_workers = min(32, os.cpu_count() + 4)

        logger.info(f"Starting multiprocessing sum calculation for {len(data)} elements")
        logger.info(f"Using {max_workers} processes")
        start_time = time.time()

        # Разделение данных на блоки
        chunks = split_data(data, self.chunk_size)

        # Подготовка данных для процессов
        chunk_data = [(chunk, i) for i, chunk in enumerate(chunks)]

        results = []

        # Использование ProcessPoolExecutor
        with ProcessPoolExecutor(max_workers=max_workers) as executor:
            # Отправка задач в пул
            future_to_chunk = {
                executor.submit(calculate_chunk_sum, chunk_data_item): i 
                for i, chunk_data_item in enumerate(chunk_data)
            }

            # Сбор результатов по мере завершения
            for future in as_completed(future_to_chunk):
                chunk_id = future_to_chunk[future]
                try:
                    result = future.result()
                    results.append(result)
                    logger.info(f"Completed chunk {chunk_id}")
                except Exception as e:
                    logger.error(f"Chunk {chunk_id} generated an exception: {e}")

        # Подсчет общей суммы
        total_sum = sum(chunk_sum for _, chunk_sum in results)

        end_time = time.time()
        execution_time = end_time - start_time

        logger.info(f"Multiprocessing calculation completed in {execution_time:.2f} seconds")
        logger.info(f"Total sum: {total_sum}")

        return total_sum

    def calculate_sum_chunked(self, data: List[int], chunk_size: int = 5, max_workers: int = None) -> int:
        """Многопроцессный подсчет блоками для лучшего контроля памяти"""
        if max_workers is None:
            max_workers = min(32, os.cpu_count() + 4)

        logger.info(f"Starting chunked multiprocessing calculation for {len(data)} elements")
        start_time = time.time()

        # Разделение данных на блоки
        chunks = split_data(data, self.chunk_size)

        # Разделение блоков на группы
        chunk_groups = [chunks[i:i + chunk_size] for i in range(0, len(chunks), chunk_size)]

        all_results = []

        with ProcessPoolExecutor(max_workers=max_workers) as executor:
            for group_id, chunk_group in enumerate(chunk_groups):
                logger.info(f"Processing group {group_id} with {len(chunk_group)} chunks")

                # Подготовка данных для группы
                group_data = [(chunk, i) for i, chunk in enumerate(chunk_group)]

                # Отправка группы задач
                future_to_chunk = {
                    executor.submit(calculate_chunk_sum, chunk_data_item): i 
                    for i, chunk_data_item in enumerate(group_data)
                }

                # Сбор результатов группы
                group_results = []
                for future in as_completed(future_to_chunk):
                    chunk_id = future_to_chunk[future]
                    try:
                        result = future.result()
                        group_results.append(result)
                    except Exception as e:
                        logger.error(f"Chunk {chunk_id} generated an exception: {e}")

                all_results.extend(group_results)

        # Подсчет общей суммы
        total_sum = sum(chunk_sum for _, chunk_sum in all_results)

        end_time = time.time()
        execution_time = end_time - start_time

        logger.info(f"Chunked multiprocessing calculation completed in {execution_time:.2f} seconds")
        logger.info(f"Total sum: {total_sum}")

        return total_sum

def calculate_sum_batch(chunk_batch: List[Tuple[List[int], int]]) -> List[Tuple[int, int]]:
    """Функция для пакетной обработки блоков"""
    results = []
    for chunk_data in chunk_batch:
        result = calculate_chunk_sum(chunk_data)
        results.append(result)
    return results

def main():
    # Создание калькулятора
    calculator = MultiprocessingSumCalculator(chunk_size=50000)

    # Генерация тестовых данных
    data_size = 1000000  # 1 миллион элементов
    data = generate_data(data_size)

    print(f"Generated {len(data)} random numbers")
    print(f"Expected sum (approximate): {len(data) * 500}")  # Среднее значение 500

    # Тестирование многопроцессного подсчета
    print("\n" + "="*50)
    print("MULTIPROCESSING CALCULATION")
    print("="*50)

    result1 = calculator.calculate_sum_multiprocessing(data, max_workers=4)

    # Тестирование блочного подсчета
    print("\n" + "="*50)
    print("CHUNKED MULTIPROCESSING CALCULATION")
    print("="*50)

    result2 = calculator.calculate_sum_chunked(data, chunk_size=3, max_workers=4)

    # Проверка результатов
    print(f"\nResults match: {result1 == result2}")
    print(f"Final sum: {result1}")

if __name__ == "__main__":
    # Важно для multiprocessing на Windows
    multiprocessing.freeze_support()
    main()

    Особенности многопроцессного подхода

    1. Истинный параллелизм - процессы выполняются на разных ядрах CPU
    2. Изоляция процессов - каждый процесс имеет свою память
    3. Высокие накладные расходы - создание процессов дороже потоков
    4. Сложность обмена данными - необходимо сериализация

    Сравнение производительности

    Скрипт для тестирования

    import time
import asyncio
import statistics
from typing import List, Dict
import psutil
import os

# Импорт калькуляторов
from sum_count_async import AsyncSumCalculator
from sum_count_threading import ThreadingSumCalculator
from sum_count_multiprocessing import MultiprocessingSumCalculator, generate_data

def measure_performance(calculator_func, data: List[int], name: str) -> Dict:
    """Измерение производительности калькулятора"""
    print(f"\nТестирование {name}...")

    # Измерение времени
    start_time = time.time()
    start_memory = psutil.Process().memory_info().rss / 1024 / 1024  # MB

    # Выполнение подсчета
    if name == "Async":
        result = asyncio.run(calculator_func(data))
    else:
        result = calculator_func(data)

    end_time = time.time()
    end_memory = psutil.Process().memory_info().rss / 1024 / 1024  # MB

    execution_time = end_time - start_time
    memory_usage = end_memory - start_memory

    return {
        "name": name,
        "execution_time": execution_time,

    Обработка исключений в процессах

    def calculate_chunk_sum_safe(chunk_data: Tuple[List[int], int]) -> Tuple[int, int]:
    """Безопасный подсчет суммы с обработкой ошибок"""
    try:
        chunk, chunk_id = chunk_data

        logger.info(f"Processing chunk {chunk_id} in process {os.getpid()}")

        # Проверка входных данных
        if not chunk:
            logger.warning(f"Empty chunk {chunk_id}")
            return chunk_id, 0

        # Подсчет суммы
        chunk_sum = sum(chunk)

        logger.info(f"Chunk {chunk_id} completed successfully")
        return chunk_id, chunk_sum

    except Exception as e:
        logger.error(f"Error processing chunk {chunk_id}: {e}")
        return chunk_id, 0

    def calculate_sum_with_retry(data: List[int], max_retries: int = 3) -> int:
    """Подсчет суммы с повторными попытками"""
    for attempt in range(max_retries):
        try:
            calculator = MultiprocessingSumCalculator(chunk_size=50000)
            return calculator.calculate_sum_multiprocessing(data, max_workers=4)
        except Exception as e:
            logger.error(f"Attempt {attempt + 1} failed: {e}")
            if attempt == max_retries - 1:
                raise
            time.sleep(2 ** attempt)  # Экспоненциальная задержка

    Заключение

    Подсчет суммы демонстрирует различия между подходами к параллельному программированию:

    • asyncio - неэффективен для CPU-интенсивных задач из-за GIL
    • threading - также ограничен GIL, но проще в использовании
    • multiprocessing - лучший выбор для CPU-интенсивных задач

    Для CPU-интенсивных операций multiprocessing показывает наилучшую производительность, несмотря на высокие накладные расходы.
