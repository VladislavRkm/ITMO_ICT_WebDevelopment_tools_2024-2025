    Практика 2: SQLModel и база данных

    Описание

    Вторая практика посвящена интеграции FastAPI с базой данных PostgreSQL через SQLModel. Мы создадим полноценное API с постоянным хранением данных, используя современный ORM SQLModel.

    Цели

    • Изучить SQLModel - современный ORM для Python
    • Настроить подключение к PostgreSQL
    • Реализовать CRUD операции с базой данных
    • Понять миграции и управление схемой БД

    Архитектура

    Приложение использует SQLModel для работы с базой данных PostgreSQL:

    practice_2/
├── connection.py    # Настройка подключения к БД
├── models.py        # SQLModel модели
└── main.py          # FastAPI приложение

    Настройка базы данных

    connection.py

    from sqlmodel import SQLModel, create_engine, Session
from typing import Generator

# Создание движка базы данных
DATABASE_URL = "postgresql://postgres:postgres@localhost:5432/book_exchange"
engine = create_engine(DATABASE_URL, echo=True)

# Создание таблиц
def create_db_and_tables():
    SQLModel.metadata.create_all(engine)

# Генератор сессий
def get_session() -> Generator[Session, None, None]:
    with Session(engine) as session:
        yield session

    Модели данных

    models.py

    from sqlmodel import SQLModel, Field, Relationship
from typing import Optional, List
from datetime import datetime

class BookBase(SQLModel):
    title: str = Field(index=True)
    author: str = Field(index=True)
    description: Optional[str] = None
    isbn: Optional[str] = Field(default=None, index=True)
    publication_year: Optional[int] = None
    condition: str = Field(default="good")

class Book(BookBase, table=True):
    id: Optional[int] = Field(default=None, primary_key=True)
    created_at: datetime = Field(default_factory=datetime.utcnow)
    updated_at: datetime = Field(default_factory=datetime.utcnow)

class BookCreate(BookBase):
    pass

class BookRead(BookBase):
    id: int
    created_at: datetime
    updated_at: datetime

class BookUpdate(SQLModel):
    title: Optional[str] = None
    author: Optional[str] = None
    description: Optional[str] = None
    isbn: Optional[str] = None
    publication_year: Optional[int] = None
    condition: Optional[str] = None

    Основное приложение

    main.py

    from fastapi import FastAPI, Depends, HTTPException
from sqlmodel import Session, select
from typing import List
from connection import engine, create_db_and_tables, get_session
from models import Book, BookCreate, BookRead, BookUpdate

app = FastAPI()

# Создание таблиц при запуске
@app.on_event("startup")
def on_startup():
    create_db_and_tables()

@app.get("/")
def read_root():
    return {"message": "Book Exchange API with Database"}

@app.post("/books/", response_model=BookRead)
def create_book(book: BookCreate, session: Session = Depends(get_session)):
    db_book = Book.from_orm(book)
    session.add(db_book)
    session.commit()
    session.refresh(db_book)
    return db_book

@app.get("/books/", response_model=List[BookRead])
def read_books(skip: int = 0, limit: int = 100, session: Session = Depends(get_session)):
    books = session.exec(select(Book).offset(skip).limit(limit)).all()
    return books

@app.get("/books/{book_id}", response_model=BookRead)
def read_book(book_id: int, session: Session = Depends(get_session)):
    book = session.get(Book, book_id)
    if book is None:
        raise HTTPException(status_code=404, detail="Book not found")
    return book

@app.put("/books/{book_id}", response_model=BookRead)
def update_book(book_id: int, book_update: BookUpdate, session: Session = Depends(get_session)):
    db_book = session.get(Book, book_id)
    if db_book is None:
        raise HTTPException(status_code=404, detail="Book not found")

    book_data = book_update.dict(exclude_unset=True)
    for key, value in book_data.items():
        setattr(db_book, key, value)

    session.add(db_book)
    session.commit()
    session.refresh(db_book)
    return db_book

@app.delete("/books/{book_id}")
def delete_book(book_id: int, session: Session = Depends(get_session)):
    book = session.get(Book, book_id)
    if book is None:
        raise HTTPException(status_code=404, detail="Book not found")

    session.delete(book)
    session.commit()
    return {"message": "Book deleted successfully"}

    API Endpoints

    POST /books/

    Создание новой книги.

    Тело запроса: 

    {
    "title": "The Great Gatsby",
    "author": "F. Scott Fitzgerald",
    "description": "A story of the Jazz Age",
    "isbn": "978-0743273565",
    "publication_year": 1925,
    "condition": "good"
}

    Ответ: Созданная книга с ID и временными метками

    GET /books/

    Получение списка всех книг с пагинацией.

    Параметры: - skip (int, опционально) - количество записей для пропуска - limit (int, опционально) - максимальное количество записей

    Ответ: Список книг

    GET /books/{book_id}

    Получение книги по ID.

    Параметры: - book_id (int) - ID книги

    Ответ: Книга или 404 ошибка

    PUT /books/{book_id}

    Обновление книги по ID.

    Параметры: - book_id (int) - ID книги

    Тело запроса: Частичное обновление (только измененные поля)

    Ответ: Обновленная книга

    DELETE /books/{book_id}

    Удаление книги по ID.

    Параметры: - book_id (int) - ID книги

    Ответ: Сообщение об успешном удалении

    Ключевые особенности SQLModel

    Преимущества

    1. Современный синтаксис - основан на Pydantic и SQLAlchemy
    2. Типизация - полная поддержка типов Python
    3. Автоматическая валидация - встроенная валидация данных
    4. Простота использования - минимальный boilerplate код
    5. Производительность - оптимизированные запросы

    Отличия от SQLAlchemy

    • Более простой синтаксис - меньше кода для типичных операций
    • Лучшая интеграция с FastAPI - автоматическая генерация схем
    • Современные возможности - поддержка новых версий Python
    • Type hints - полная поддержка типизации

    SQLModel автоматически валидирует данные на основе типов и ограничений:

    class BookBase(SQLModel):
    title: str = Field(min_length=1, max_length=200)
    author: str = Field(min_length=1, max_length=100)
    publication_year: Optional[int] = Field(ge=1800, le=2024)

    Миграции

    Ручные миграции

    # Создание новой таблицы
def create_new_table():
    SQLModel.metadata.create_all(engine, tables=[NewTable.__table__])

# Удаление таблицы
def drop_table():
    NewTable.__table__.drop(engine)

    Автоматические миграции

    SQLModel автоматически создает таблицы при запуске приложения:

    @app.on_event("startup")
def on_startup():
    create_db_and_tables()
