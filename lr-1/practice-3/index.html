    Практика 3: Миграции с Alembic

    Описание

    Третья практика посвящена изучению системы миграций Alembic для управления схемой базы данных. Мы научимся создавать, применять и откатывать миграции, что является важным навыком для разработки и развертывания приложений.

    Цели

    • Изучить Alembic - систему миграций для SQLAlchemy
    • Научиться создавать и применять миграции
    • Понять управление версиями схемы БД
    • Освоить откат миграций

    Архитектура

    Приложение использует Alembic для управления миграциями:

    migrations/env.py

    from logging.config import fileConfig
from sqlalchemy import engine_from_config
from sqlalchemy import pool
from alembic import context
from models import SQLModel

config = context.config

if config.config_file_name is not None:
    fileConfig(config.config_file_name)

target_metadata = SQLModel.metadata

def run_migrations_offline() -> None:
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )

    with context.begin_transaction():
        context.run_migrations()

def run_migrations_online() -> None:
    connectable = engine_from_config(
        config.get_section(config.config_ini_section, {}),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )

    with connectable.connect() as connection:
        context.configure(
            connection=connection, target_metadata=target_metadata
        )

        with context.begin_transaction():
            context.run_migrations()

if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()

    Модели данных

    models.py

    from sqlmodel import SQLModel, Field, Relationship
from typing import Optional, List
from datetime import datetime

class Skill(SQLModel, table=True):
    id: Optional[int] = Field(default=None, primary_key=True)
    name: str = Field(unique=True, index=True)
    description: Optional[str] = None
    level: int = Field(default=1, ge=1, le=10)
    created_at: datetime = Field(default_factory=datetime.utcnow)

class User(SQLModel, table=True):
    id: Optional[int] = Field(default=None, primary_key=True)
    username: str = Field(unique=True, index=True)
    email: str = Field(unique=True, index=True)
    full_name: Optional[str] = None
    is_active: bool = Field(default=True)
    created_at: datetime = Field(default_factory=datetime.utcnow)

class UserSkill(SQLModel, table=True):
    id: Optional[int] = Field(default=None, primary_key=True)
    user_id: Optional[int] = Field(default=None, foreign_key="user.id")
    skill_id: Optional[int] = Field(default=None, foreign_key="skill.id")
    proficiency_level: int = Field(default=1, ge=1, le=5)
    acquired_at: datetime = Field(default_factory=datetime.utcnow)

    Создание миграций

    Инициализация Alembic

alembic init migrations

    Создание первой миграции

    # Создание миграции на основе текущих моделей
alembic revision --autogenerate -m "Initial migration"

    Применение миграций

    # Применение всех миграций
alembic upgrade head

# Применение до конкретной версии
alembic upgrade 805327bf0ca6

# Применение одной миграции вперед
alembic upgrade +1

    Откат миграций

    # Откат на одну миграцию назад
alembic downgrade -1

# Откат до конкретной версии
alembic downgrade 805327bf0ca6

# Полный откат
alembic downgrade base

    Миграция с данными

    """add default skills

Revision ID: 123456789abc
Revises: 805327bf0ca6
Create Date: 2024-01-15 11:00:00.000000

"""
from alembic import op
import sqlalchemy as sa

# revision identifiers, used by Alembic.
revision = '123456789abc'
down_revision = '805327bf0ca6'
branch_labels = None
depends_on = None

def upgrade() -> None:
    # Добавление данных
    op.execute("""
        INSERT INTO skill (name, description, level, created_at) VALUES
        ('Python', 'Programming language', 1, NOW()),
        ('FastAPI', 'Web framework', 1, NOW()),
        ('SQLAlchemy', 'ORM library', 1, NOW()),
        ('PostgreSQL', 'Database', 1, NOW())
    """)

def downgrade() -> None:
    # Удаление данных
    op.execute("DELETE FROM skill WHERE name IN ('Python', 'FastAPI', 'SQLAlchemy', 'PostgreSQL')")

from alembic import op
import sqlalchemy as sa

# revision identifiers, used by Alembic.
revision = '456789abcdef'
down_revision = '123456789abc'
branch_labels = None
depends_on = None

def upgrade() -> None:
    # Обновление данных
    op.execute("UPDATE skill SET level = 2 WHERE name = 'Python'")
    op.execute("UPDATE skill SET level = 3 WHERE name = 'FastAPI'")

def downgrade() -> None:
    # Откат изменений
    op.execute("UPDATE skill SET level = 1 WHERE name IN ('Python', 'FastAPI')")

    Структура миграций

    1. Одна миграция - одно изменение - не объединяйте несколько изменений
    2. Обратимость - всегда пишите функцию downgrade()
    3. Тестирование - тестируйте миграции на копии БД
    4. Документирование - описывайте сложные изменения

    Мониторинг миграций

    Отслеживание состояния

    from alembic.script import ScriptDirectory
from alembic.runtime.migration import MigrationContext
from sqlalchemy import create_engine

def get_migration_status():
    """Получение статуса миграций"""
    engine = create_engine("postgresql://...")

    with engine.connect() as conn:
        context = MigrationContext.configure(conn)
        current_rev = context.get_current_revision()

        script = ScriptDirectory.from_config(Config("alembic.ini"))
        head_rev = script.get_current_head()

        return {
            "current": current_rev,
            "head": head_rev,
            "behind": current_rev != head_rev
        }
